cmake_minimum_required(VERSION 3.16)
project(monotrack_court LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# Prefer a normal development installation (e.g. Homebrew OpenCV on macOS).
find_package(OpenCV 4 QUIET COMPONENTS core imgproc imgcodecs calib3d highgui videoio)
if(NOT OpenCV_FOUND)
  include(FetchContent)
  set(BUILD_LIST core,imgproc,imgcodecs,calib3d,highgui,videoio CACHE STRING "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_PERF_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_opencv_python2 OFF CACHE BOOL "" FORCE)
  set(BUILD_opencv_python3 OFF CACHE BOOL "" FORCE)
  set(BUILD_opencv_java OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(WITH_FFMPEG OFF CACHE BOOL "" FORCE)
  set(WITH_GSTREAMER OFF CACHE BOOL "" FORCE)
  set(WITH_GTK OFF CACHE BOOL "" FORCE)
  set(WITH_QT OFF CACHE BOOL "" FORCE)
  set(WITH_IPP OFF CACHE BOOL "" FORCE)
  set(OPENCV_GENERATE_PKGCONFIG OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(opencv
    URL https://github.com/opencv/opencv/archive/refs/tags/4.12.0.tar.gz
    URL_HASH SHA256=44c106d5bb47efec04e531fd93008b3fcd1d27138985c5baf4eafac0e1ec9e9d)
  FetchContent_MakeAvailable(opencv)
  list(APPEND OpenCV_INCLUDE_DIRS ${opencv_SOURCE_DIR}/include ${opencv_BINARY_DIR})
  set(OpenCV_LIBS opencv_core opencv_imgproc opencv_imgcodecs opencv_calib3d opencv_highgui opencv_videoio)
  foreach(module IN LISTS OpenCV_LIBS)
    get_target_property(module_includes ${module} INCLUDE_DIRECTORIES)
    list(APPEND OpenCV_INCLUDE_DIRS ${module_includes})
  endforeach()
endif()

file(GLOB SOURCES CONFIGURE_DEPENDS "*.cpp")
add_executable(monotrack-court ${SOURCES})
target_include_directories(monotrack-court PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(monotrack-court PRIVATE ${OpenCV_LIBS})
